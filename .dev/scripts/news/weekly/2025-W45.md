```dataview
TABLE WITHOUT ID
  coalesce(title_ja, title) AS "タイトル",
  link(file.link, "記事ページへ") AS "記事ページへ",
  link(source_url, "引用元 ↗") AS "引用元"
FROM "news/articles"
WHERE date >= date("2025-11-03") AND date <= date("2025-11-09")
SORT date DESC
```
